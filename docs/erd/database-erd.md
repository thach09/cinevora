    USERS {
        bigint id PK "GENERATED BY DEFAULT AS IDENTITY"
        varchar username "UNIQUE, NOT NULL (Max 50)"
        varchar email "UNIQUE, NOT NULL (Max 100)"
        varchar password "NOT NULL (BCrypt hash, Max 255)"
        varchar role "NOT NULL (ADMIN, CUSTOMER)"
        boolean is_deleted "NOT NULL DEFAULT FALSE"
        timestamp created_at "NOT NULL DEFAULT CURRENT_TIMESTAMP"
        timestamp updated_at "NOT NULL DEFAULT CURRENT_TIMESTAMP"
    }

    CUSTOMERS {
        bigint user_id PK, FK "One-to-One với USERS.id"
        varchar membership_type "NOT NULL DEFAULT 'FREE'"
        timestamp registration_date "NOT NULL DEFAULT CURRENT_TIMESTAMP"
    }

    CATEGORIES {
        bigint id PK "GENERATED BY DEFAULT AS IDENTITY"
        varchar name "UNIQUE, NOT NULL (Max 100)"
        text description
        boolean is_deleted "NOT NULL DEFAULT FALSE"
        timestamp created_at "NOT NULL DEFAULT CURRENT_TIMESTAMP"
        timestamp updated_at "NOT NULL DEFAULT CURRENT_TIMESTAMP"
    }

    MOVIES {
        bigint id PK "GENERATED BY DEFAULT AS IDENTITY"
        varchar title "NOT NULL (Max 255)"
        text description
        integer duration_minutes "NOT NULL"
        varchar video_url "NOT NULL"
        varchar thumbnail_url
        integer release_year
        integer views "NOT NULL DEFAULT 0"
        boolean is_deleted "NOT NULL DEFAULT FALSE"
        timestamp created_at "NOT NULL DEFAULT CURRENT_TIMESTAMP"
        timestamp updated_at "NOT NULL DEFAULT CURRENT_TIMESTAMP"
    }

    MOVIE_CATEGORIES {
        bigint movie_id PK, FK
        bigint category_id PK, FK
    }

    WATCHLIST {
        bigint customer_id PK, FK
        bigint movie_id PK, FK
        timestamp added_at "NOT NULL DEFAULT CURRENT_TIMESTAMP"
    }

    FAVOURITES {
        bigint customer_id PK, FK
        bigint movie_id PK, FK
        timestamp added_at "NOT NULL DEFAULT CURRENT_TIMESTAMP"
    }

    WATCH_PROGRESS {
        bigint id PK "GENERATED BY DEFAULT AS IDENTITY"
        bigint customer_id FK "NOT NULL"
        bigint movie_id FK "NOT NULL"
        integer last_position_seconds "NOT NULL DEFAULT 0"
        boolean is_completed "NOT NULL DEFAULT FALSE"
        timestamp updated_at "NOT NULL DEFAULT CURRENT_TIMESTAMP"
    }
